/* Experiment 4: Quick Sort (Divide and Conquer)
   Choose a pivot, place it at its correct position so that smaller
   elements are on its left and larger on its right, then sort both
   sides recursively.
   Time: O(n log n) best/average, O(n^2) worst. Space: O(log n). */
#include <stdio.h>

int partition(int a[], int low, int high) {
    int pivot = a[low];                  /* first element as pivot */
    int i = low + 1, j = high, temp;

    while (1) {
        while (i <= high && a[i] <= pivot)
            i++;                         /* move right till bigger element */
        while (a[j] > pivot)
            j--;                         /* move left till smaller/equal element */
        if (i < j) {
            temp = a[i];
            a[i] = a[j];
            a[j] = temp;
        } else
            break;
    }
    temp = a[low];                       /* put pivot in its final place */
    a[low] = a[j];
    a[j] = temp;
    return j;
}

void quickSort(int a[], int low, int high) {
    int p;
    if (low < high) {
        p = partition(a, low, high);
        quickSort(a, low, p - 1);        /* sort left of pivot */
        quickSort(a, p + 1, high);       /* sort right of pivot */
    }
}

int main() {
    int a[100], n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    quickSort(a, 0, n - 1);

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\n");
    return 0;
}
