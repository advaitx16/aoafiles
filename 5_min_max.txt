/* Experiment 5: Finding Maximum and Minimum (Divide and Conquer)
   Divide the array into two halves, find min and max of each half
   recursively, then compare the two results.
   Comparisons: about 3n/2 - 2 (fewer than 2n - 2 of the simple method).
   Time: O(n). */
#include <stdio.h>

int max, min;

void maxMin(int a[], int low, int high) {
    int mid, max1, min1;

    if (low == high) {                   /* only one element */
        max = min = a[low];
    }
    else if (high == low + 1) {          /* only two elements */
        if (a[low] > a[high]) {
            max = a[low];
            min = a[high];
        } else {
            max = a[high];
            min = a[low];
        }
    }
    else {
        mid = (low + high) / 2;
        maxMin(a, low, mid);             /* left half */
        max1 = max;
        min1 = min;
        maxMin(a, mid + 1, high);        /* right half */
        if (max1 > max) max = max1;      /* combine results */
        if (min1 < min) min = min1;
    }
}

int main() {
    int a[100], n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    maxMin(a, 0, n - 1);

    printf("Maximum element = %d\n", max);
    printf("Minimum element = %d\n", min);
    return 0;
}
